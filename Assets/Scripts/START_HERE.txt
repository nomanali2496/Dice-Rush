================================================================================
                          🔥 COMMON COMMANDS
================================================================================

// Check if logged in
bool loggedIn = PlayFabManager.Instance.IsLoggedIn();

// Get player info
string name = PlayFabManager.Instance.currentPlayerData.playerName;
int cash = PlayFabManager.Instance.currentPlayerData.cash;

// Add currency
PlayFabManager.Instance.AddCash(100);
PlayFabManager.Instance.AddCoins(50);

// Spend currency
if (PlayFabManager.Instance.SpendCash(100)) {
    Debug.Log("Purchase successful!");
}

// Save data (auto-saves on currency changes)
PlayFabManager.Instance.SavePlayerData();

Fix: Check GamePanel assigned to SplashScreen script

Problem: Can't type
Fix: Use TextMeshPro InputFields, check EventSystem exists
